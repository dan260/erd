            REFERENCE
            
    Schema syntax
    
          
    
          
        
    

        
    
          
    
            The whole schema is described in text. The preview and the
            relations are rebuilt as you type.
          
    

          
    
            
    
              
    
                entity Name { … }
                
    An entity and its fields. Each field — on a new line.
    
              
    
              
    
                field or
                  field: type
                
    The type is optional. A line with a colon is just more familiar.
    
              
    
              
    
                author: User
                
    
                  If the type matches an entity name, the field becomes a relation.
                
    
              
    
              
    
                posts: Post[1:n]
                
    
                  Explicit cardinality [owner:target]: 1 — one, n — many.
                
    
              
    
              
    
                author: User[n:1?]
                
    
                  ? adds zero: 1? — 0 or 1, n? — zero to infinity.
                
    
              
    
              
    
                # comment
                
    
                  Lines that start with # or // are ignored.
                
    
              
    
              
    
                rel Post[1:n]
                
    
                  Declares a relation inside the entity without creating a field.
                  Read as [owner:target], the cardinality is required.
                
    
              
    
            
    

            
    
              
    
                example
                a relation from both sides
              
    
              
    entity User {
  id: int
  name
  posts: Post[1:n]
}

entity Post {
  id: int
  title: string
  author: User[n:1]
}
    
              
    
                User.posts and
                Post.author describe the same one-to-many
                relation and will be merged.
